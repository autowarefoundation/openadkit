[Pod]
PodName=awf-oak-x5h
# Host networking, matching deployments/samples/scenario-simulation. Domain 1
# discovery then stays on loopback, and a container sharing this namespace
# can see tap0 -- a host interface created by rpmsg-eth.service, not
# something a rootless pod network could reach.
Network=host

# MEMBERSHIP: this pod holds the SEVEN core Autoware components only --
# map, planning, vehicle, system, control, simulator, api. awf-oak-bridge
# and awf-oak-scenario are deliberately NOT members, and must not be
# "reunified" back in.
#
# Why: Quadlet gives the generated pod service a `Wants=` on every member,
# and gives every member a `BindsTo=` on the pod. Starting any single
# component therefore starts the pod, and the pod then starts every other
# member. With all nine in the pod, `systemctl start awf-oak-map` also
# started awf-oak-bridge and awf-oak-scenario -- observed on the board,
# 2026-08-18 -- which makes "Autoware alone on domain 1", the entire premise
# of the Stage 1 isolation gate, unreachable by any choice of units an
# operator can make. x5h-stack-smoke.sh's `autoware` mode asserts
# `bridge_active` is false precisely so a Stage 1 PASS cannot be collected on
# a bridged board; with the bridge in the pod that assertion fires on every
# run. The scenario runner is the same problem in a worse form: it has no
# `[Install]` section specifically so a scenario never runs at boot, and pod
# membership defeated that (observed: awf-oak-scenario killed with SIGTERM
# after being started by nobody).
#
# What the two ex-members lose by leaving, and how they get it back: pod
# membership here conveyed only the shared network namespace, because
# Network=host above means that namespace IS the host's. Both units
# therefore carry their own `Network=host` in [Container], which is exactly
# equivalent -- the bridge still sees tap0 for domain 2 and both still reach
# domain 1 on loopback. To be exhaustive about the rest, since the point of
# this note is what leaving the pod costs: the pod declares no Volume= and no
# EnvironmentFile= that reaches members; the ipc/uts namespaces a podman pod
# shares are unused here (CycloneDDS is configured for UDP, not shared memory
# -- see cyclonedds-x5h.xml -- and USE_POINTCLOUD_CONTAINER is false); and
# the one remaining thing membership conveys, the pod cgroup as the members'
# cgroup parent (podman's --share-parent, default true), has no consequence
# on this profile -- the pod sets no resource limits, and Quadlet emits
# --cgroups=split for every container unit regardless, so systemd owns each
# container's cgroup either way.
#
# Three things -- not two -- are what let any container in this stack start
# at all, member of this pod or not. All nine container units carry ALL of:
#
#   1. EnvironmentFile=/etc/containers/systemd/awf-oak-x5h.env in [Service],
#      so *systemd* can expand the ${...} references in the ExecStart=
#      Quadlet generates (Image=${API_IMAGE} and friends). Quadlet renames
#      the source [Container] section to [X-Container], which systemd
#      ignores, so the [Container] copy below does NOT reach systemd. This
#      is the one that is easy to lose, precisely because sitting next to
#      the [Container] copy it looks redundant: delete it and every
#      ${...} in ExecStart expands to the empty string, podman exits with
#      "repository name must have at least one component", and every
#      unit fails at once -- which the board smoke reports as a plain
#      unit_inactive, giving no hint that an env file is missing.
#   2. EnvironmentFile=/etc/containers/systemd/awf-oak-x5h.env in
#      [Container], which Quadlet turns into `podman run --env-file`, so the
#      container *process* sees CYCLONEDDS_URI, ROS_DOMAIN_ID and the rest.
#      systemd's own environment is not passed into the container.
#   3. Volume=/etc/containers/systemd/cyclonedds-x5h.xml:/autoware/cyclonedds.xml:ro,z,
#      the file CYCLONEDDS_URI points at.
#
# Items 2 and 3 are mandatory, not belt-and-braces: the component images
# ship their own /home/aw/cyclonedds.xml requesting
# <SocketReceiveBufferSize min="10MB"/>, while net.core.rmem_max is 212992
# bytes on this board. Under the image's own config, CycloneDDS aborts at
# participant creation ("failed to increase socket receive buffer size to
# at least 10485760 bytes, current is 425984 bytes" -> "error creating
# node: rcl node's rmw handle is invalid" -> exit 1). The env file's
# CYCLONEDDS_URI=file:///autoware/cyclonedds.xml (applied over the image's
# ENV by podman) plus that volume are what avoid it.

# Two keys from deployments/samples/scenario-simulation/docker-compose.yaml
# are deliberately NOT reproduced here: `init: true` and the map service's
# `pid: service:map`. A podman pod shares ipc, net and uts namespaces but
# not pid, so `pid: service:map` has no Quadlet equivalent at pod level --
# and both are inert on this profile anyway, because
# USE_POINTCLOUD_CONTAINER=false means nothing joins the map container's
# process namespace. If a future scenario turns that on, PodmanArgs=--share=...
# on the pod (or per-container) is the lever, not a silent re-derivation of
# the compose file.
